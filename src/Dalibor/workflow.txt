
1)	# Follow workflow on https://cran.r-project.org/web/packages/snpsettest/vignettes/reference_1000Genomes.html
	# Download hg37 data from https://www.cog-genomics.org/plink/2.0/resources#1kg_phase3

        # create folder workdir/input and go thither

2)	plink2 --zst-decompress all_phase3.pgen.zst > all_phase3.pgen

3)	qsub plink_job.sge
	# this selects the correct population (EUR) and generates subpopulations
	# it also transforms the output to Plink1 format

4)	python make_bed.py
	# create bed files from plink

5)	# concatenate all .bed files using `cat` command to out/EUR_phase3_chrALL.bed

6)	python gwas_associations.py
	# this generates gwas_associations_LDblock.csv

7)	qsub make_gwas_snp_in_grb.sge
	# this runs gwas_snps_in_grb.R or gwas_snps_in_grb_no_kurtosis.R
	# you need to select kurtosis or no_kurtosis by changing the script name in .sge script

8)	# create directories for plots
	mkdir plots

	mkdir plots/gviz_plots

9)	qsub make_gviz.sge

10)	qsub make plots.sge
